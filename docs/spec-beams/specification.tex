\begin{frame}{Reliability}
    \begin{itemize}
        \item Sensor code uses UDP without any guarantee of reliabilty
        \item In some cases reliable message transport service is needed 
            \begin{itemize}
                \item to keep the common view of the sensor state synchronized
                among several clients
                \item because the client takes actions based on the received
                messages e.g. "When OFF device update is received stop the
                gas flow"
            \end{itemize}
        \item In some cases the reliability creates only unnecessary overhead
            \begin{itemize}
                \item when the message is soon outdated e.g. data streaming
            \end{itemize}
        \item We use \textit{a weak realiable message transport} which
        guarantees the delivery of the most recent message
            \begin{itemize}
                \item Can be switched on / off with a parameter given in
                subscribe message
            \end{itemize}

    \end{itemize}
\end{frame}

\begin{frame}{Protocol Encoding}
    \begin{itemize}
        \item JSON for Server and Publisher communication
        \item BSON for Subscriber and Server communication 
        \item The most throughput consuming data is binary, hence
              binary encoding
        \item Original data is in JSON, so binary representation of JSON
     \end{itemize}
\end{frame}

\begin{frame}{Message format}
  \begin{itemize}
    \item \texttt{\{'type': integer, 'sids': [\{'id': string, 'reliable': boolean\}, ...]\}}
    \item \texttt{type} - message type, integer. E.g. get sensor list - 0, sensor list reply - 1.
    \item \texttt{sids} - sensor id(s), string (list of objects).
    \item \texttt{data} - update data, depends on the sensor type. E.g. boolean - for device sensors.
    \item \texttt{time} - update timestamp.
    \item \texttt{sensor\_type} - the type of a sensor, integer.
    \item \texttt{reliable} - boolean. In a subscribe message - request for reliable delivery.
                                       In an update message - request to acknowledge the update.
    %\item \texttt{\{'type': integer, 'sid': string, 'sensor\_type': integer, 'time':
    %                integer, 'reliable': boolean, 'data': sensor\_data\}}
    %\item \texttt{\{'type': integer, 'sid': string, 'sensor\_type': integer, 'time': integer\}}
  \end{itemize}
\end{frame}

\begin{frame}{Keep alive messaging}
  \begin{itemize}
     \item In order to prune "dead" clients
     \item The client must send a message every keep-alive timeout
     \item After several timeouts and no message received the server
           deletes the client from the list
     \item The client has to re-subscribe
  \end{itemize}
\end{frame}
