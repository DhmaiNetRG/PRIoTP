\begin{frame}{Open issues}
    \begin{itemize}
        \item If versions do not match, maybe negotiate versions at the beginning,
              so backward compatibility is sustained
        \item Fragmentation
        \begin{itemize}
            \item Add fields for fragment offset and total number of fragments
            \item If a fragment is lost and the updates are delivered in
            a reliable manner the server timer will timeout and cause a
            retransmission
            \item If a fragment is lost and the updates are delivered in an
            unreliable manner the next update for the sensor will overwrite the
            data in fragmentation buffer
        \end{itemize}
    \end{itemize}
\end{frame}
